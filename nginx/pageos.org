# /etc/nginx/sites-available/pageos.org
# PageOS production — serves pageos.org and www.pageos.org (www → apex redirect).
#
# Enable:  ln -s /etc/nginx/sites-available/pageos.org /etc/nginx/sites-enabled/
# SSL:     certbot --nginx -d pageos.org -d www.pageos.org

# ── HTTP: ACME challenge + redirect ────────────────────────────────────────────

server {
    listen 80;
    server_name pageos.org www.pageos.org;

    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }

    location / {
        return 301 https://pageos.org$request_uri;
    }
}

# ── HTTPS: www → apex redirect ─────────────────────────────────────────────────

server {
    listen 443 ssl http2;
    server_name www.pageos.org;

    ssl_certificate     /etc/letsencrypt/live/pageos.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/pageos.org/privkey.pem;
    include             /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam         /etc/letsencrypt/ssl-dhparams.pem;

    return 301 https://pageos.org$request_uri;
}

# ── HTTPS: main app ─────────────────────────────────────────────────────────────

server {
    listen 443 ssl http2;
    server_name pageos.org;

    ssl_certificate     /etc/letsencrypt/live/pageos.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/pageos.org/privkey.pem;
    include             /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam         /etc/letsencrypt/ssl-dhparams.pem;

    # Security headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Content-Type-Options nosniff;
    add_header X-Frame-Options SAMEORIGIN;

    # Allow large file uploads (documents, assets)
    client_max_body_size 50M;

    # ── Go API (/api/v1/...) ────────────────────────────────────────────────────
    location /api/ {
        proxy_pass         http://127.0.0.1:8090;
        proxy_http_version 1.1;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_read_timeout 120s;
    }

    # ── Next.js web app ─────────────────────────────────────────────────────────
    location / {
        proxy_pass         http://127.0.0.1:3090;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade    $http_upgrade;
        proxy_set_header   Connection "upgrade";
        proxy_set_header   Host       $host;
        proxy_set_header   X-Real-IP  $remote_addr;
        proxy_set_header   X-Forwarded-Proto $scheme;
    }

    # Next.js static assets — long cache
    location /_next/static/ {
        proxy_pass http://127.0.0.1:3090;
        expires    1y;
        add_header Cache-Control "public, immutable";
    }
}
