# /etc/nginx/sites-available/pageos.org
# PageOS production — pageos.org + www.pageos.org
#
# Enable:  ln -sf /etc/nginx/sites-available/pageos.org /etc/nginx/sites-enabled/pageos.org
# SSL:     certbot --nginx -d pageos.org -d www.pageos.org  (adds HTTPS blocks automatically)

server {
    listen 80;
    server_name pageos.org www.pageos.org;

    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }

    # www → apex redirect (HTTP)
    if ($host = www.pageos.org) {
        return 301 http://pageos.org$request_uri;
    }

    # ── Go API (/api/v1/...) ────────────────────────────────────────────────────
    location /api/ {
        proxy_pass         http://127.0.0.1:8090;
        proxy_http_version 1.1;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_read_timeout 120s;
    }

    # ── Next.js web app ─────────────────────────────────────────────────────────
    location / {
        proxy_pass         http://127.0.0.1:3090;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade    $http_upgrade;
        proxy_set_header   Connection "upgrade";
        proxy_set_header   Host       $host;
        proxy_set_header   X-Real-IP  $remote_addr;
        proxy_set_header   X-Forwarded-Proto $scheme;
    }

    # Next.js static assets — long cache
    location /_next/static/ {
        proxy_pass http://127.0.0.1:3090;
        expires    1y;
        add_header Cache-Control "public, immutable";
    }
}
